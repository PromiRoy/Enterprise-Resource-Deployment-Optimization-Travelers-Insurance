#!/usr/bin/env python3
"""
ca5_zoo_v1.py — Cini & Abela replication, FINAL package: feature concatenation
+ their model zoo + forward-chained validation. Produces the replication's
results table.
Features per their Fig. 6 (LIWC absent, disclosed): CA-1 arc statistics + NRC
emotion/Plutchik features + arc-cluster one-hot; CA-2 crew power/experience +
collaboration-graph centralities; CA-4 keyword vector (50); CA-3 embedding
(per config: none / MiniLM-384 / UAE-1024); metadata = air index + air year
(their release-year analog; era-entangled by construction, disclosed). The
DCT-100 arc vector itself is not fed (matches their design: stats + cluster
represent the arc).
Validation: forward-chained by season — train seasons <= k, test season k+1,
k = 3..8 (their rolling-window principle at episode scale). Predictions pooled
across folds; metrics R2 / RMSE / MAE / MAPE / One-Away (their set).
Baselines: Base = training-mean (theirs; scored -0.03 on their catalog) and
drift-ridge meta (air index + air year; season one-hots cannot extrapolate to
an unseen season, so the honest metadata baseline is the time trend).
Models (their zoo, library defaults per their own no-tuning disclosure):
XGBoost, LightGBM, SVR, Lasso, CatBoost, ElasticNet, MLP. Full zoo on the
primary config (UAE + crew); SVR (their best) across all six configs.
Also: their Low/Med/High bins printed as the degeneracy demo.
Installs (once): pip install xgboost lightgbm catboost
Run: nohup python3 -u ca5_zoo_v1.txt > ca5_out.txt 2>&1 &  ; tail -f ca5_out.txt
"""
import os
for v in ("OMP_NUM_THREADS", "OPENBLAS_NUM_THREADS", "MKL_NUM_THREADS"):
    os.environ[v] = "2"
import re, warnings; warnings.filterwarnings("ignore")
import numpy as np
import pandas as pd

def P(*a): print(*a, flush=True)

man = pd.read_csv("manifest.csv")
cols = {c.lower(): c for c in man.columns}
inc = None
for cand in ("included", "include", "is_included", "excluded", "exclude"):
    if cand in cols:
        v = man[cols[cand]]
        b = v.values.astype(bool) if v.dtype == bool else v.astype(str).str.lower().isin(["true", "1", "yes"]).values
        inc = ~b if "exclud" in cand else b
        break
assert inc is not None and inc.sum() == 151
idc = next(c for c in man.columns if man[c].astype(str).str.contains("SE0").any())
c_r = cols.get("rating") or cols.get("imdb_rating")
sf_ids = man.loc[inc, idc].astype(str).tolist()
y = man.loc[inc, c_r].astype(float).values
season = np.array([int(re.search(r"SE(\d\d)", s).group(1)) for s in sf_ids])

vw = pd.read_csv("seinfeld_nielsen_viewership.csv")[["sf_id", "airdate"]]
vw["base"] = vw["sf_id"].str.extract(r"(SF_SE\d\d_EP\d\d)")[0]
amap = dict(zip(vw["base"], pd.to_datetime(vw["airdate"])))
airdt = pd.Series([amap.get(re.search(r"SF_SE\d\d_EP\d\d", s).group(0)) for s in sf_ids])
air_year = airdt.dt.year.values.astype(float)
air_index = airdt.rank(method="first").values.astype(float)

ca1 = pd.read_csv("features_ca1.csv")
assert list(ca1.sf_id) == sf_ids
clus = pd.get_dummies(ca1.pop("arc_cluster"), prefix="arc_clus").values.astype(float)
ca1_X = ca1.drop(columns=["sf_id"]).values.astype(float)
ca2 = pd.read_csv("features_ca2.csv")
assert list(ca2.sf_id) == sf_ids
ca2_X = ca2.drop(columns=["sf_id"]).values.astype(float)
kw = np.load("features_ca4_keywords.npy")
emb = {"none": np.zeros((151, 0)),
       "minilm": np.load("features_ca3_minilm.npy"),
       "uae": np.load("features_ca3_uae.npy")}
meta = np.column_stack([air_index, air_year])

def stack(emb_tag, crew):
    parts = [ca1_X, clus, kw, emb[emb_tag], meta] + ([ca2_X] if crew else [])
    return np.hstack(parts)

from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.linear_model import Ridge, Lasso, ElasticNet
from sklearn.svm import SVR
from sklearn.neural_network import MLPRegressor
from xgboost import XGBRegressor
from lightgbm import LGBMRegressor
from catboost import CatBoostRegressor
from sklearn.metrics import r2_score, mean_absolute_error, mean_squared_error

def zoo():
    return {
        "XGBoost": XGBRegressor(n_estimators=300, max_depth=3, learning_rate=0.05,
                                random_state=0, verbosity=0),
        "LightGBM": LGBMRegressor(n_estimators=300, max_depth=3, learning_rate=0.05,
                                  random_state=0, verbose=-1),
        "SVR": SVR(),
        "Lasso": Lasso(alpha=0.1),
        "CatBoost": CatBoostRegressor(iterations=300, depth=3, learning_rate=0.05,
                                      random_seed=0, verbose=0),
        "ElasticNet": ElasticNet(alpha=0.1),
        "MLP": MLPRegressor(hidden_layer_sizes=(64,), early_stopping=True,
                            max_iter=2000, random_state=0),
    }

FOLDS = [(k, k + 1) for k in range(3, 9)]   # train <=k, test k+1

def run(model_maker, X):
    pred, truth = [], []
    for k, t in FOLDS:
        tr, te = np.where(season <= k)[0], np.where(season == t)[0]
        if len(te) == 0: continue
        pipe = Pipeline([("imp", SimpleImputer(strategy="mean")),
                         ("sc", StandardScaler()), ("m", model_maker())])
        pipe.fit(X[tr], y[tr])
        pred.append(pipe.predict(X[te])); truth.append(y[te])
    p, t = np.concatenate(pred), np.concatenate(truth)
    mape = float(np.mean(np.abs((t - p) / t)))
    return {"R2": r2_score(t, p), "RMSE": float(np.sqrt(mean_squared_error(t, p))),
            "MAE": mean_absolute_error(t, p), "MAPE": mape,
            "OneAway": float(np.mean(np.abs(t - p) <= 1.0))}

def show(name, m):
    P(f"{name:26s} R2={m['R2']:7.3f}  RMSE={m['RMSE']:.3f}  MAE={m['MAE']:.3f}  "
      f"MAPE={m['MAPE']:.4f}  OneAway={m['OneAway']:.3f}")

P(f"n={len(y)}  test rows (S4-S9) = {int((season>=4).sum())}  y sd={y.std():.3f}")
P("\n=== BASELINES (forward-chained) ===")
class Mean:
    def fit(self, X, yy): self.m = yy.mean(); return self
    def predict(self, X): return np.full(len(X), self.m)
show("Base (train mean)", run(lambda: Mean(), np.zeros((151, 1))))
show("meta drift-ridge", run(lambda: Ridge(alpha=1.0), meta))

P("\n=== FULL ZOO — primary config: UAE embedding + crew ===")
Xp = stack("uae", True)
for name, m in zoo().items():
    show(name, run((lambda mm=m: mm.__class__(**mm.get_params())) if not isinstance(m, CatBoostRegressor)
                   else (lambda: CatBoostRegressor(iterations=300, depth=3, learning_rate=0.05,
                                                   random_seed=0, verbose=0)), Xp))

P("\n=== SVR (their best model) across all six configs ===")
for et in ("none", "minilm", "uae"):
    for crew in (True, False):
        show(f"SVR {et:6s} {'with' if crew else 'no'}-crew", run(lambda: SVR(), stack(et, crew)))

P("\n=== their Low/Med/High bins (Table 1) on Seinfeld ===")
bins = np.digitize(y, [4.0, 7.0])
u, c = np.unique(bins, return_counts=True)
P(f"class counts (0-4 / 4.1-7 / 7.1+): {dict(zip(u.tolist(), c.tolist()))}"
  f"  -> all episodes in one class; categorisation degenerate at within-show scale.")
P("\nDONE. This table + the CA-1..CA-4 notes = the replication section.")
