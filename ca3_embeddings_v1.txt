#!/usr/bin/env python3
"""
ca3_embeddings_v1.py — Cini & Abela replication, Package 3: script embeddings
with THEIR end-weighted pooling (Section 3.3.3).
Two-model scope (their smallest + their best):
  * all-MiniLM-L6-v2      — 256-token chunks, runs on CPU
  * stella_en_1.5B_v5     — 512-token chunks, needs the GPU (~4-6GB, fp16,
                            trust_remote_code; downloads ~3GB on first run)
Pooling: linearly increasing weights across a document's chunks (w_i ~ i+1,
normalized) so the ending carries the highest weight — their Peak-End design.
Input: cleaned noemo/ episode texts, speaker labels KEPT (their scripts
contain character names; only the keyword package stripped them).
Output: features_ca3_minilm.npy + features_ca3_stella.npy (manifest order)
        + a mean pairwise-cosine line per model (between-episode contrast).
CHECK nvidia-smi FIRST. If the GPU is busy, run anyway: stella falls back to
CPU with a warning (slow but correct), or rerun later.
Run: nohup python3 -u ca3_embeddings_v1.txt > ca3_out.txt 2>&1 &  ; tail -f ca3_out.txt
"""
import os
for v in ("OMP_NUM_THREADS", "OPENBLAS_NUM_THREADS", "MKL_NUM_THREADS"):
    os.environ[v] = "4"
import csv, json, re
import numpy as np
import pandas as pd

def P(*a): print(*a, flush=True)

man = pd.read_csv("manifest.csv")
cols = {c.lower(): c for c in man.columns}
inc = None
for cand in ("included", "include", "is_included", "excluded", "exclude"):
    if cand in cols:
        v = man[cols[cand]]
        b = v.values.astype(bool) if v.dtype == bool else v.astype(str).str.lower().isin(["true", "1", "yes"]).values
        inc = ~b if "exclud" in cand else b
        break
assert inc is not None and inc.sum() == 151
idc = next(c for c in man.columns if man[c].astype(str).str.contains("SE0").any())
sf_ids = man.loc[inc, idc].astype(str).tolist()

REMOVE = set()
if os.path.exists("emocue_decisions.csv"):
    for row in csv.DictReader(open("emocue_decisions.csv")):
        if row["bucket"].strip().upper() == "REMOVE":
            REMOVE.add(row["cue"].strip())
BR = re.compile(r"\[([^\[\]]+)\]")

def clean(text):
    def sub(m):
        s = m.group(1).strip()
        return "" if (s in REMOVE or s == "Scene ends") else m.group(0)
    t = BR.sub(sub, text)
    t = re.sub(r"^L\d+\.\s*", "", t, flags=re.M)
    return re.sub(r"\n{3,}", "\n\n", t)

def episode_text(sf):
    base = re.search(r"SF_SE\d\d_EP\d\d", sf).group(0)
    if base == "SF_SE04_EP04": base = "SF_SE04_EP03"
    se, ep = int(base[5:7]), int(base[10:12])
    if se == 1:
        p = f"pilot/chunks_E{ep:02d}.json"
    else:
        p = f"reasoning_run/noemo/chunks_{base}.json"
        if not os.path.exists(p): p = f"reasoning_run/chunks_{base}.json"
    d = json.load(open(p))
    return clean("\n".join(c["text"] for c in (d["chunks"] if isinstance(d, dict) else d)))

def word_chunks(text, n_words):
    w = text.split()
    return [" ".join(w[i:i + n_words]) for i in range(0, len(w), n_words)] or [""]

def end_weighted(M):
    w = np.arange(1, len(M) + 1, dtype=float); w /= w.sum()
    return (M * w[:, None]).sum(0)

def mean_pairwise_cos(X):
    Xn = X / (np.linalg.norm(X, axis=1, keepdims=True) + 1e-12)
    C = Xn @ Xn.T
    return float((C.sum() - len(X)) / (len(X) * (len(X) - 1)))

texts = [episode_text(sf) for sf in sf_ids]
from sentence_transformers import SentenceTransformer
import torch
dev = "cuda" if torch.cuda.is_available() else "cpu"
P(f"device for the large model: {dev}")

JOBS = [("minilm", "sentence-transformers/all-MiniLM-L6-v2", 200, "cpu", {}),
        ("stella", "NovaSearch/stella_en_1.5B_v5", 400, dev,
         {"trust_remote_code": True})]
for tag, name, n_words, device, kw in JOBS:
    P(f"\n=== {tag}: {name} (chunks of ~{n_words} words, device={device}) ===")
    model = SentenceTransformer(name, device=device, **kw)
    if device == "cuda":
        try: model = model.half()
        except Exception: pass
    V = []
    for i, t in enumerate(texts):
        ch = word_chunks(t, n_words)
        E = model.encode(ch, batch_size=16, show_progress_bar=False,
                         convert_to_numpy=True, normalize_embeddings=False)
        V.append(end_weighted(np.asarray(E, dtype=np.float32)))
        if (i + 1) % 25 == 0: P(f"  {i+1}/151 episodes embedded")
    V = np.vstack(V)
    np.save(f"features_ca3_{tag}.npy", V)
    P(f"features_ca3_{tag}.npy {V.shape}  mean pairwise cosine = {mean_pairwise_cos(V):.3f}")
    del model
    if device == "cuda": torch.cuda.empty_cache()
P("\nDONE. Next: CA-5 (concatenate all feature blocks -> their model zoo, forward-chained).")
